<script>
const ETAPAS = ["Novo Lead","1º Contato","Qualificação","Proposta / Lance","Negociação","Venda / Contemplado","Pós-venda"];
let leads = JSON.parse(localStorage.getItem('pipe_lucas')||'[]');
let meta = parseInt(localStorage.getItem('meta_lucas')||'3');
let filtro = '';
let rotinaSalva = JSON.parse(localStorage.getItem('rotina_lucas')||'[]');
document.getElementById('metaInput').value = meta;
document.querySelectorAll('#rotina input').forEach((c,i)=>{ c.checked = !!rotinaSalva[i]; });

function salvar(){ localStorage.setItem('pipe_lucas', JSON.stringify(leads)); }
function salvarRotina(){
  const v=[]; document.querySelectorAll('#rotina input').forEach(c=>v.push(c.checked));
  localStorage.setItem('rotina_lucas', JSON.stringify(v));
}
function setMeta(){ meta = parseInt(document.getElementById('metaInput').value||'3'); localStorage.setItem('meta_lucas', meta); render(); }
function hoje(){ return new Date().toISOString().slice(0,10); }
function dias(d){ if(!d) return 99; return Math.floor((new Date(hoje())-new Date(d))/(1000*60*60*24)); }
function addLead(){
  const nome=document.getElementById('nome').value.trim(), fone=document.getElementById('fone').value.trim();
  if(!nome||!fone){ alert('Preencha nome e WhatsApp'); return; }
  leads.unshift({id:Date.now(), nome, fone, valor:parseInt(document.getElementById('valor').value||'0'), etapa:document.getElementById('etapa').value, temp:document.getElementById('temp').value, grupo:document.getElementById('grupo').value, obj:document.getElementById('obj').value, obs:document.getElementById('obs').value, data:hoje(), falouHoje:false});
  salvar(); ['nome','fone','valor','grupo','obj','obs'].forEach(id=>document.getElementById(id).value=''); render();
}
function filtrar(f){ filtro=f; render(); }
function mover(id){
  const l=leads.find(x=>x.id===id); const i=ETAPAS.indexOf(l.etapa);
  if(i<ETAPAS.length-1) l.etapa=ETAPAS[i+1]; salvar(); render();
}
function voltar(id){
  const l=leads.find(x=>x.id===id); const i=ETAPAS.indexOf(l.etapa);
  if(i>0) l.etapa=ETAPAS[i-1]; salvar(); render();
}
function falar(id){
  const l=leads.find(x=>x.id===id);
  const txt=encodeURIComponent(`Olá ${l.nome}! ✨ Aqui é o Lucas Fortes - Consórcio Porto Seguro. Vi que você tem interesse em ${l.obj||'consórcio'} ${l.grupo?`(grupo ${l.grupo})`:''}. Tenho uma condição especial pra você hoje, posso te mostrar?`);
  window.open(`https://wa.me/55${l.fone.replace(/\D/g,'')}?text=${txt}`,'_blank');
  l.falouHoje=true; l.data=hoje(); salvar(); render();
}
function excluir(id){ if(confirm('Excluir lead?')){ leads=leads.filter(x=>x.id!==id); salvar(); render(); } }
function render(){
  const busca=(document.getElementById('busca').value||'').toLowerCase();
  let lista=leads.filter(l=>{
    if(busca && !l.nome.toLowerCase().includes(busca)) return false;
    if(filtro==='Quente' && l.temp!=='Quente') return false;
    if(filtro==='3dias' && dias(l.data)<3) return false;
    if(filtro==='hoje' && l.falouHoje) return false;
    return true;
  });
  const vendas=leads.filter(l=>l.etapa==='Venda / Contemplado').length;
  document.getElementById('vendasHoje').textContent=vendas;
  document.getElementById('metaTxt').textContent=meta;
  document.getElementById('barra').style.width=Math.min(100, vendas/meta*100)+'%';
  document.getElementById('valorPipe').textContent='R$ '+leads.reduce((s,l)=>s+(l.valor||0),0).toLocaleString('pt-BR');
  document.getElementById('faltaFalar').textContent=leads.filter(l=>!l.falouHoje).length;
  const pri=leads.find(l=>l.temp==='Quente' && dias(l.data)>=3) || leads.find(l=>l.temp==='Quente') || leads[0];
  document.getElementById('prioridade').textContent=pri?pri.nome:'Tudo em dia 🎉';

  document.getElementById('board').innerHTML=ETAPAS.map(et=>{
    const itens=lista.filter(l=>l.etapa===et);
    return `<div class="col"><h3>${et}</h3><p>${itens.length} leads • ${et==='Novo Lead'?'Falar em até 5 min':et==='Negociação'?'Foco em fechar hoje':et==='Pós-venda'?'Pedir indicação':''}</p>
    ${itens.map(l=>`<div class="lead">
      <b>${l.falouHoje?'✅ ':'❌ '}${l.nome}</b><br><small>📱 ${l.fone} ${l.grupo?'• '+l.grupo:''}<br>💰 R$ ${(l.valor||0).toLocaleString('pt-BR')} • ${l.obj||''}<br>${dias(l.data)===0?'Hoje':dias(l.data)+' dia(s) sem falar'}</small>
      <br><span class="tag ${l.temp}">${l.temp}</span>
      ${l.obs?`<div style="font-size:13px;background:#F8FAFC;padding:6px;border-radius:8px">${l.obs}</div>`:''}
      <div class="actions"><button class="wa" onclick="falar(${l.id})">💬 WhatsApp</button><button class="next" onclick="mover(${l.id})">➡️</button><button class="del" onclick="excluir(${l.id})">🗑️</button></div>
      <button onclick="voltar(${l.id})" style="background:#F1F5F9;width:100%;margin-top:6px;font-size:12px">⬅️ Voltar etapa</button>
    </div>`).join('') || '<p style="text-align:center;padding:10px">Nenhum aqui</p>'}
    </div>`
  }).join('');
}
render();
</script>
